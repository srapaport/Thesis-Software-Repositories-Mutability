\begin{frame}[fragile,squeeze]{March 2025: a tag that changed meaning}

    \begingroup\footnotesize
    \begin{itemize}\setlength{\itemsep}{0.05em}
        \item \textbf{GitHub Actions} are CI/CD steps that each \texttt{uses:} a
              third-party action pinned by a \textbf{tag} --- like
              \texttt{tj-actions/changed-files}.
        \item Attackers retargeted \AttackTagMovesCount{} existing versions
              (\texttt{v39}\dots\texttt{v47}) to \textbf{one malicious commit}.\srcmark
        \item \AttackStableCases{} of those tags had been stable
              $\geq\AttackStableDaysLowerBound{}$ days --- trusted pins, not noise.
        \item Anyone resolving the name then \textbf{ran attacker code}, leaking
              developer credentials.
    \end{itemize}
    \endgroup
    \vspace{0.15em}
    \begin{terminalboxsmall}
$ # your workflow, unchanged:
$ uses: tj-actions/changed-files@v45
$ # same name (*$\Rightarrow$*) different code after the next fetch
    \end{terminalboxsmall}
    \vspace{0.15em}
    {\footnotesize No exploit --- an ordinary Git update. So how does a published name change?}
    \srcnote{Palo Alto Unit~42, \emph{GitHub Actions Supply Chain Attack}, 2025.
    \url{https://unit42.paloaltonetworks.com/github-actions-supply-chain-attack/}}
\end{frame}

% Problem: the incident is not a measurement. The block is the question list

\begin{frame}{What we need to know about tag alterations}
    \begin{block}{Questions}
        \begin{enumerate}\setlength{\itemsep}{0.1em}
            \item \textbf{Detection.} Can we detect an alteration at scale?
            \item \textbf{Prevalence.} How common is it, and who does it?
            \item \textbf{Impact.} What changes, and can we observe the consequences?
        \end{enumerate}
    \end{block}
    %\vspace{0.6em}
    \begingroup\small
    \begin{itemize}\setlength{\itemsep}{0.55em}
        \item \textbf{Scale.} \OriginsWithTagsShort{} origins with at least one tag;
              \TotalTagsShort{} tags in total.
        \item \textbf{What is missing.} The map \emph{tag name $\rightarrow$ previous
              target} --- the only thing that would make an alteration visible.
        \item \textbf{What ``altered'' can mean.} The name \textbf{disappears}; it is \textbf{moved};
              or it comes \textbf{back} after a deletion.
    \end{itemize}
    \endgroup
\end{frame}

\begin{frame}{What we need to know about rewritten histories}
    \begin{block}{Questions}
        \begin{enumerate}\setlength{\itemsep}{0.1em}
            \item \textbf{Detection.} Can we detect a rewritten history at scale?
            \item \textbf{Prevalence.} How common is it, and who does it?
            \item \textbf{Impact.} What changes, and can we observe the consequences?
        \end{enumerate}
    \end{block}
    \begingroup\small
    \begin{itemize}\setlength{\itemsep}{0.55em}
        \item \textbf{Scale.} \OriginsTwoVisitsOrMoreShort{} Git origins visited at

              \SWHRevisionsShort{} revisions.
        \item \textbf{What is missing.} The previous reachable graph. A rewritten
              commit is not deleted; it becomes \emph{unreachable}, and the forge
              stops advertising it.

              versus every descendant rewritten with it (\textbf{snowball}). Counting
              commits without that\\split is meaningless.

\section{Secret Removal in Version Control Archives}
%##############################################################################

% ------------------------------------------------------------------
\begin{frame}{Removal is not revocation}
    \begingroup
    \begin{itemize}\setlength{\itemsep}{1em}
        \item Hard-coded credentials leak at scale and are a long-standing weakness class.
        \item After a leak, two actions are \textbf{not interchangeable}:
              \begin{itemize}
                  \item \textbf{Cleanup} (optional) --- rewrite the repository so the
                        current view no longer shows the file. \emph{Visible, easy.}
                  \item \textbf{Rotation} (mandatory) --- revoke the credential at the
                        provider that issued it. \emph{The only thing that invalidates
                        a leak.}
              \end{itemize}
        \item Cleanup is easy to mistake for rotation. An archive cannot see rotation ---
              but it \textbf{can} show whether erased secrets remain \textbf{recoverable}.
    \end{itemize}
    \endgroup
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Three conditions that reinforce integrity}

    Three minimal conditions to \emph{reinforce} integrity, rather than assume it:
    \vspace{0.4em}
    \begin{center}
    \begin{tikzpicture}[
        c/.style={draw=tptred, line width=0.8pt, rounded corners=2pt, align=left,

        \node[c] (c1) {\textbf{1.\ Preserve evidence at the mutation boundary.}
            Forges can log reference create/retarget/delete/force-push in append-only logs before prior targets vanish.};
        \node[c, below=6pt of c1] (c2) {\textbf{2.\ Bind consumption to verifiable identity.}
            A reproducible build/audit should retain and verify the resolved content identifier, not just a name.};
        \node[c, below=6pt of c2] (c3) {\textbf{3.\ Close incidents at the layer of authority.}
            Fix each incident where its effects live: build records, audit evidence, provider-side revocation.};

\begin{frame}[fragile]{Backup --- Tier C predefined network checks}

    authorisation.\par\endgroup

# SSH private key (GitHub): active iff banner "Hi <user>!"

# GitHub PAT: active iff HTTP 200
curl -H "Authorization: Bearer ghp_..." https://api.github.com/user

# AWS access key: active iff account ARN returned
aws sts get-caller-identity
